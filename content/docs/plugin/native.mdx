C 与 C++ 用**同一份头文件** `fuyubot_plugin.h`，差别只有两条：导出符号要不要放进 `extern "C"`，
以及用什么拼 JSON。头文件是**产物**，不是源码——它由 cbindgen 从 Rust 定义生成，
用 `fuyubot-sdk codegen` 落盘，别手改（改了下次就被覆盖）。

<Callout type="info">
  C / C++ / Go 没有 async 运行时：请求与更新用回调（`on`），工具结果用
  `call_tool_result` 同步拿。可 await 的版本只在 Rust / JS / TS / Python 的脚本 SDK 里，
  见 [异步等待](./async.mdx)。
</Callout>

```bash
fuyubot-sdk init ping --lang c --dir .        # 生成 ping.c + fuyubot_plugin.h
fuyubot-sdk init ping --lang cpp --dir .      # 生成 plugin.cpp + fuyubot_plugin.h
fuyubot-sdk codegen --lang c --dir <工程目录>   # 只刷新头文件（C++ 用 --lang cpp，同一份）
```

## 七个符号

v3 是四件套，v4 又加了三个（都可以不写，但写了才有对应能力）。

| 符号 | 必须吗 | 说明 |
| --- | --- | --- |
| `plugin_init` | **必须** | 比 `host->version != ABI_VERSION`，存下 VTable，返回 `PluginHandle` |
| `plugin_start` | 可选 | 订阅/注册；返回 `0` 表示成功 |
| `plugin_stop` | 可选 | 清状态；只登记元数据的插件可以不写 |
| `plugin_free_string` | **必须** | 释放本插件 `malloc` 出来的字符串（`plugin_manifest` 返回的那块也走它） |
| `plugin_manifest` | v4 可选 | 返回自描述 JSON（名字 / 版本 / `inject` / `soft_inject`），宿主在 `init` 之前读 |
| `fuyubot_abi_version` | v4 可选 | 静态返回 `ABI_VERSION`，`fuyubot-sdk check` 用它判断版本 |
| `plugin_free_handle` | v4 可选 | ABI 版本不匹配时宿主靠它回收 `init` 分配的状态（那时不能调 `plugin_stop`） |

`PluginHandle` 的判据是「`abi_version != 0` **且** `context` 非空」，所以 `context` 不能给 `NULL`。
版本不对就返回全零句柄，宿主会把它当成拒绝加载。

## C

```c
#include "fuyubot_plugin.h"

static const HostApiVTable *g_host = NULL;
static int g_state = 0;              /* PluginHandle::context 必须非空 */

/* 回调与宿主在同一根线程上跑；ABI 是能 unwind 的那种（对 C 而言就是普通 C 调用）。 */
static int32_t on_update(void *user_data, const char *payload) {
    (void)user_data;
    /* 判正文 == "/ping"，然后： */
    g_host->td_invoke_simplified(g_host->host_ctx, "sendMessage", params);
    g_host->log(g_host->host_ctx, 3, "ping 插件：回了 pong");
    return 0;
}

FUYUBOT_EXPORT PluginHandle plugin_init(const HostApiVTable *host) {
    PluginHandle failed = {0, 0, NULL, NULL};
    if (host == NULL || host->version != ABI_VERSION) return failed;
    g_host = host;
    PluginHandle handle = { ABI_VERSION, (uint32_t)sizeof(PluginHandle), &g_state, "ping" };
    return handle;
}

FUYUBOT_EXPORT int32_t plugin_start(PluginHandle handle, const char *config_json) {
    (void)config_json;
    if (handle.context == NULL || g_host == NULL) return -1;
    SubscriptionHandle s =
        g_host->on(g_host->host_ctx, "updateNewMessage", NULL, on_update, NULL);
    return s.id == 0 ? -1 : 0;
}

FUYUBOT_EXPORT void plugin_stop(PluginHandle handle) { (void)handle; g_host = NULL; }
FUYUBOT_EXPORT void plugin_free_string(char *ptr) { (void)ptr; }
```

`ABI_VERSION` 与 `FUYUBOT_EXPORT` 都来自头文件，不要自己写死。

## C++

同 C，**唯一的硬差别**是七个导出符号必须在 `extern "C"` 里——C++ 会改函数名（mangling），
宿主按 ABI 定死的名字 `dlsym`，改了就找不到。

```cpp
extern "C" {

FUYUBOT_EXPORT PluginHandle plugin_init(const HostApiVTable *host) {
    PluginHandle failed{0, 0, nullptr, nullptr};
    if (host == nullptr || host->version != ABI_VERSION) return failed;
    g_host = host;
    return PluginHandle{ ABI_VERSION, (uint32_t)sizeof(PluginHandle), &g_state, "ping" };
}

FUYUBOT_EXPORT int32_t plugin_start(PluginHandle handle, const char *config_json);
FUYUBOT_EXPORT void plugin_stop(PluginHandle handle);
FUYUBOT_EXPORT void plugin_free_string(char *ptr);

}  // extern "C"
```

<Callout type="warn">
  `C-unwind` 对 C/C++ 而言与普通 C 调用**ABI 一致**，但 C++ 的异常照样能穿过这个边界——
  逃到宿主那边的异常是未定义行为。**在回调内部把异常处理完再返回**。
</Callout>

## 常用 ABI 怎么调（C / C++ 同一套）

每个函数第一个参数都是 `g_host->host_ctx`。下面把常用的走了一遍：

```c
/* ① 订阅（第二个参数是过滤器 JSON，NULL = 全收）/ 退订 */
SubscriptionHandle s = g_host->on(g_host->host_ctx, "updateNewMessage",
                                  "{\"message.content.text.text\":\"/ping\"}", on_update, NULL);
g_host->off(g_host->host_ctx, s);

/* ② 发 TDLib 请求：只拿请求号，响应走 "response" 通道 */
RequestId id = g_host->td_invoke_simplified(g_host->host_ctx, "sendMessage", params);

/* ③ 同步执行不碰网络的方法：当场返回，发消息不能走它 */
const char *parsed = g_host->td_execute(g_host->host_ctx,
    "{\"@type\":\"parseTextEntities\",\"text\":\"*粗体*\","
    "\"parse_mode\":{\"@type\":\"textParseModeMarkdown\",\"version\":2}}");

/* ④ 发自定义事件：自己 + 全局订阅者收得到 */
g_host->emit(g_host->host_ctx, "ping.seen", "{\"chat_id\":123}");

/* ⑤ 打日志：Error=1 Warn=2 Info=3 Debug=4 Trace=5（非法值按 Info） */
g_host->log(g_host->host_ctx, 3, "回了 pong");

/* ⑥ 读任意配置段 / 写自己那段（写成功会回写 fuyubot.toml） */
const char *city = g_host->get_config(g_host->host_ctx, "my.weather", "city");
int ok = g_host->set_config(g_host->host_ctx, "last_seen", "123");

/* ⑦ 登记工具（handler 给 NULL = 只登记元数据）/ 调别人的工具 / 列工具 */
ToolHandle t = g_host->register_tool(g_host->host_ctx, descriptor_json, NULL, NULL);
const char *out = g_host->call_tool_result(g_host->host_ctx, "weather.now", "{\"city\":\"上海\"}");
const char *tools = g_host->list_tools(g_host->host_ctx);

/* ⑧ 挂中间件：阶段 0..5（见「工具与中间件」） */
MiddlewareHandle m = g_host->use_middleware(g_host->host_ctx, 2, on_stage, NULL);

/* ⑨ 阻塞等一条匹配的更新；timeout_ms == 0 表示只检查当前状态 */
WaitHandle w = g_host->wait_for_update(g_host->host_ctx, "{\"@type\":\"updateNewMessage\"}", 5000);

/* ⑩ 服务：注册到自己作用域 / 注册到进程级 / 沿父链查 */
ServiceHandle svc = g_host->provide_service(g_host->host_ctx, "demo.cache", &vtable);
const ServiceVTable *got = g_host->get_service(g_host->host_ctx, "demo.cache");

/* ⑪ v5 简化入口：发一条 op（异步，结果照 "response" 通道回投）/ 同步拿不碰网络的结果 */
RequestId rid = g_host->simplify_invoke(g_host->host_ctx,
    "{\"op\":\"send_message\","
     "\"args\":{\"chat_id\":-1001234567890,"
              "\"input_message_content\":{\"@type\":\"inputMessageText\",\"text\":\"hi\"}}}");
const char *r = g_host->simplify_invoke_result(g_host->host_ctx,
    "{\"op\":\"parse_text_entities\","
     "\"args\":{\"text\":\"**b**\",\"parse_mode\":\"markdown_v2\"}}");
/* r 同上归宿主线程局部缓冲，读完即弃，不要 free */
```

⚠️ 三条所有权规则（跨界最容易出事的地方）：

- `td_execute` / `get_config` / `call_tool_result` / `list_tools` / `list_plugins` 返回的
  `const char *` 是**宿主线程局部缓冲**，下一次宿主调用之前有效——要留就当场 `strdup`，
  **不要 free 它**。
- 反过来，只有 `plugin_manifest` 返回的那块是**你自己 malloc 的**，宿主读完用
  `plugin_free_string` 还给你（所以必须真能 free，别返回静态字符串）。
- `on` 的 `filter_json` 里 `event_type` 键只用来**选通道**，不参与负载匹配。

## 编译（你自己的工具链）

```bash
# MSVC：先 call vcvars64.bat；/utf-8 不能省（源码与头文件都是 UTF-8）
cl /nologo /utf-8 /LD /I <工程目录> <工程目录>/ping.c /Fe:<工程目录>/target/debug/ping.dll

# gcc / clang
cc -shared -fPIC -I <工程目录> <工程目录>/ping.c -o <工程目录>/target/debug/ping.dll
g++ -shared -fPIC -I <工程目录> <工程目录>/plugin.cpp -o <工程目录>/target/debug/ping.dll
```

`/utf-8` 省掉的后果很难看：中文注释被按系统代码页拆坏，编译器报的是「注释提前结束、后面全是语法错误」。

编完装进插件目录：

```bash
fuyubot plugin install <工程目录>
```

## 读 / 写配置

每个插件在 `fuyubot.toml` 的 `[plugin.<插件名>]` 下有自己的一段配置（见[配置文件](../usage/config.mdx)）。
C 侧直接调 VTable 上那两个函数：

```c
/* 读自己的段 `[plugin.ping]` 里的 greeting；返回 JSON 文本，没有就是 NULL。 */
const char *greeting = g_host->get_config(g_host, "ping", "greeting");
if (greeting != NULL) {
    /* greeting 指向 "…"（带引号的 JSON 字符串）；要用就当场拷走。 */
    printf("greeting = %s\n", greeting);
}

/* 读别人的段（只读）：section 是那个插件的名字。 */
const char *city = g_host->get_config(g_host, "my.weather", "city");

/* 写自己的段：只能写自己（没有 section 参数），成功返回 0，并回写 fuyubot.toml。 */
int32_t code = g_host->set_config(g_host, "last_started", "\"2026-01-01\"");
if (code != 0) { /* 越权 / 解析失败 / 写盘失败 */ }
```

两点与其它函数一致：**返回值是宿主线程局部缓冲**（读完即弃，别 free）；`set_config`
**没有 section 参数**——宿主按调用者身份判定，所以插件无法伪造写别人的段。

## 手写 `sendMessage`（或者用 `simplify_invoke`）

> ✅ **ABI v5 起，这一节只是「另一条路」，不再是唯一的路。** 宿主从 v5 开始自己跑
> 同一份 `fuyubot-simplify` 代码，经 `simplify_invoke` / `simplify_invoke_result` 暴露成一张
> **op 表**（`{"op":"send_message","args":{…}}`），所有语言都能用——发消息等结局、超时撤回、
> bot 与用户号挑不同反应方法、Markdown → entities、富消息本地块树，都不用手写。
> 见 [ABI › simplify 操作表](./abi.mdx#simplify-操作表)。
> C / C++ **不需要** SDK 薄封装——直接调 VTable 上的 `simplify_invoke` /
> `simplify_invoke_result`（见下），或者照下面手写 `sendMessage`。


先来一个能跑的：直接调 VTable 上的 `simplify_invoke` 发消息——宿主替你解析文本、等到真发出去、
超时先撤回；结果经已有的 `"response"` 事件回投（带 `request_id`，与你收 TDLib 响应是同一个 `on` 通道）：

```c
/* 异步发消息：返回 RequestId，结果照 "response" 通道回到 on_update（request_id 配对） */
RequestId rid = g_host->simplify_invoke(g_host->host_ctx,
    "{\"op\":\"send_message\","
     "\"args\":{\"chat_id\":-1001234567890,"
              "\"input_message_content\":{\"@type\":\"inputMessageText\","
                                          "\"text\":\"你好，这是用 simplify 发的\"},"
              "\"timeout_ms\":30000}}");

/* 同步拿不碰网络的结果（parse_text_entities / rich_from_markdown / list_ops）：
   返回宿主线程局部缓冲，读完即弃，不要 free */
const char *r = g_host->simplify_invoke_result(g_host->host_ctx,
    "{\"op\":\"parse_text_entities\","
     "\"args\":{\"text\":\"**粗体**\",\"parse_mode\":\"markdown_v2\"}}");
/* r 形如 {"@type":"simplifyResult","op":"parse_text_entities","ok":true,"result":{...}} */
```

`on_update` 里按 `request_id == rid` 配对，就能从那条 `response` 事件里取出 `simplifyResult` JSON
（和 `td_invoke_simplified` 的响应是同一个 `on` 通道，逻辑完全复用）。

手写 `sendMessage` 是**另一条路**——下面这些字段你都得自己拼对：

| 想要的效果 | 手写的关键字段 |
| --- | --- |
| 纯文本 | `input_message_content` = `inputMessageText`，`text` 是 `formattedText`（`text` + `entities`） |
| Markdown / HTML | **不自己解析**：先 `td_execute` 调 `parseTextEntities`（`version: 2` = MarkdownV2），把返回的 `formattedText` 原样塞进去 |
| 回复某条 | `reply_to` = `{"@type": "inputMessageReplyToMessage", "message_id": N}` |
| 发到话题 | `topic_id` = `{"@type": "messageTopicForum", "forum_topic_id": N}` |
| 内联按钮 | `reply_markup` = `replyMarkupInlineKeyboard`；callback 的 `data` 是 **base64** |
| 静音 / 保护内容 | 在 `options`（`messageSendOptions`）里，不是顶层 |
| 等到「真发出去」 | 请求号只到**临时消息**；要最终 `Message` 得再等 `updateMessageSendSucceeded` |

```c
static void send_text(long long chat_id, const char *text) {
    /* 示例用 snprintf 拼：正文里有引号 / 反斜杠就要转义，真项目请引一个 JSON 库。 */
    char params[512];
    snprintf(params, sizeof(params),
             "{\"chat_id\":%lld,\"input_message_content\":{\"@type\":\"inputMessageText\","
             "\"text\":{\"@type\":\"formattedText\",\"text\":\"%s\",\"entities\":[]}}}",
             chat_id, text);
    g_host->td_invoke_simplified(g_host->host_ctx, "sendMessage", params);
}
```

要 Markdown 就分两步：先 `td_execute` 拿 `parseTextEntities` 的结果（一个 `formattedText`），
把它的 `text` 与 `entities` 拼进上面的 `input_message_content`。方法名拿不准时用
`fuyubot-sdk td-method sendMessage` 查真签名。

## 注意

- 模板里的 JSON 取值是**朴素查找**（`strstr`），不处理转义、不处理同名嵌套键；认真写就引一个 JSON 库。
- 宿主返回的字符串是**线程局部缓冲**，只在下一次宿主调用之前有效——当场拷走，并且**不要试图释放它**。
- `set_config` 回写是**全量重写** `fuyubot.toml`，手写注释会丢（见上面「读 / 写配置」）。
- 想少拼 JSON，可以只发 `td_invoke_simplified` 需要的参数对象（`@type` 由宿主补），见 [调用 TDLib](./tdlib.mdx)。
